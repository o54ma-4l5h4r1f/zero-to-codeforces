-- number-theory | Break n into prime factors, small to big, with repeats: 12 gives 2, 2, 3. O(√n). What is left above 1 at the end is one more prime.
>>> global factorize
vector<long long> factorize(long long n) {
    vector<long long> f;
    for (long long p = 2; p * p <= n; p++)
        while (n % p == 0) {
            f.push_back(p);
            n /= p;
        }
    if (n > 1) f.push_back(n);
    return f;
}

== mod | s2 | Modulo and counting formulas
-- modular | Rules for answers modulo 10^9+7, with a and b already in 0 .. m-1. Take % after every step. Never divide normally under mod.
add: {{mod_add: (a + b) % m}}
subtract: {{mod_sub: ((a - b) % m + m) % m}}
multiply: {{mod_mul: (1LL * a * b) % m}}
divide: $$mod_div: a / b \to a \cdot b^{m-2} \bmod m$$ ($$: m$$ prime)
-- modular | Fast power: a^b mod m in O(log b) steps by squaring. Needs m below about 3·10^9 so that the products fit in long long.
>>> global power
long long power(long long a, long long b,
                long long m) {
    long long r = 1 % m; a %= m;
    while (b > 0) {
        if (b & 1) r = r * a % m;
        a = a * a % m;
        b >>= 1;
    }
    return r;
}
<<<
-- combinatorics | Counting formulas: ways to order, ways to choose, number of pairs, number of subsets.
$$fact_def: n! = 1 \cdot 2 \cdot \dots \cdot n$$
$$ncr_def: C(n, r) = \frac{n!}{r! \cdot (n-r)!}$$
$$ncr_pascal: C(n, r) = C(n-1, r-1) + C(n-1, r)$$
pairs: $$ncr_pairs: \frac{n(n-1)}{2}$$ · subsets: $$ncr_subsets: 2^n$$
-- combinatorics span2 | nCr mod a prime, many times: precompute factorials and inverse factorials once with init(), then each answer is O(1).
>>> global ncr
const int MOD = 1e9 + 7, MAXN = 200005;
long long fact[MAXN], inv[MAXN];
long long pw(long long a, long long b) {
    long long r = 1; a %= MOD;
    for (; b > 0; b >>= 1, a = a * a % MOD)
        if (b & 1) r = r * a % MOD;
    return r;
}
void init() {
    fact[0] = 1;
    for (int i = 1; i < MAXN; i++) fact[i] = fact[i-1] * i % MOD;
    inv[MAXN-1] = pw(fact[MAXN-1], MOD - 2);
    for (int i = MAXN - 1; i > 0; i--) inv[i-1] = inv[i] * i % MOD;
}
long long C(int n, int r) {
    if (r < 0 || r > n) return 0;
    return fact[n] * inv[r] % MOD * inv[n-r] % MOD;
}

-- graphs-bfs-dfs | BFS: shortest distance (number of edges) from s to every vertex. -1 means not reachable.
>>> global bfs
vector<int> bfs(int s,
        const vector<vector<int>> &g) {
    vector<int> dist(g.size(), -1);
    queue<int> q; q.push(s); dist[s] = 0;
    while (!q.empty()) {
        int v = q.front(); q.pop();
        for (int u : g[v])
            if (dist[u] == -1) {
                dist[u] = dist[v] + 1;
                q.push(u);
            }
    }
    return dist;
}

-- dijkstra span2 | Dijkstra: shortest path when edges have weights >= 0. O((n + m) log n). A distance equal to INF means not reachable.
>>> global dijkstra
const long long INF = 1e18;
vector<long long> dijkstra(int s,
        const vector<vector<pair<int,long long>>> &g) {
    vector<long long> dist(g.size(), INF);
    priority_queue<pair<long long,int>,
        vector<pair<long long,int>>, greater<>> pq;
    dist[s] = 0; pq.push({0, s});
    while (!pq.empty()) {
        auto [d, v] = pq.top(); pq.pop();
        if (d > dist[v]) continue;
        for (auto [u, w] : g[v])
            if (dist[v] + w < dist[u]) {
                dist[u] = dist[v] + w;
                pq.push({dist[u], u});
            }
    }
    return dist;
}

-- dsu | DSU: merge groups and ask if two things are in the same group. Almost O(1) per call. Same group: find(a) == find(b).
>>> global dsu
struct DSU {
    vector<int> p, sz;
    DSU(int n) : p(n + 1), sz(n + 1, 1) {
        iota(p.begin(), p.end(), 0);
    }
    int find(int x) {
        if (p[x] != x) p[x] = find(p[x]);
        return p[x];
    }
    bool unite(int a, int b) {
        a = find(a); b = find(b);
        if (a == b) return false;
        if (sz[a] < sz[b]) swap(a, b);
        p[b] = a; sz[a] += sz[b];
        return true;
    }
};

-- toposort span2 | Topological sort (Kahn): an order where every edge goes forward. If the order has fewer than n vertices, there is a cycle.
>>> global topo
vector<int> topo(int n, const vector<vector<int>> &g) {
    vector<int> in(n + 1, 0), order;
    for (int v = 1; v <= n; v++) for (int u : g[v]) in[u]++;
    queue<int> q;
    for (int v = 1; v <= n; v++) if (in[v] == 0) q.push(v);
    while (!q.empty()) {
        int v = q.front(); q.pop();
        order.push_back(v);
        for (int u : g[v]) if (--in[u] == 0) q.push(u);
    }
    return order;          // size < n  =>  cycle
}
