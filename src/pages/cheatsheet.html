long long fact[MAXN], inv[MAXN];
long long pw(long long a, long long b) {
    long long r = 1; a %= MOD;
    for (; b > 0; b >>= 1, a = a * a % MOD)
        if (b & 1) r = r * a % MOD;
    return r;
}
void init() {
    fact[0] = 1;
    for (int i = 1; i < MAXN; i++) fact[i] = fact[i-1] * i % MOD;
    inv[MAXN-1] = pw(fact[MAXN-1], MOD - 2);
    for (int i = MAXN - 1; i > 0; i--) inv[i-1] = inv[i] * i % MOD;
}
long long C(int n, int r) {
    if (r < 0 || r > n) return 0;
    return fact[n] * inv[r] % MOD * inv[n-r] % MOD;
}

vector<int> bfs(int s, const vector<vector<int>> &g) {
    vector<int> dist(g.size(), -1);
    queue<int> q; q.push(s); dist[s] = 0;
    while (!q.empty()) {
        int v = q.front(); q.pop();
        for (int u : g[v]) if (dist[u] == -1) {
            dist[u] = dist[v] + 1;
            q.push(u);
        }
    }
    return dist;
}

const long long INF = 1e18;
vector<long long> dijkstra(int s,
        const vector<vector<pair<int,long long>>> &g) {
    vector<long long> dist(g.size(), INF);
    priority_queue<pair<long long,int>,
        vector<pair<long long,int>>, greater<>> pq;
    dist[s] = 0; pq.push({0, s});
    while (!pq.empty()) {
        auto [d, v] = pq.top(); pq.pop();
        if (d > dist[v]) continue;
        for (auto [u, w] : g[v])
            if (dist[v] + w < dist[u]) {
                dist[u] = dist[v] + w;
                pq.push({dist[u], u});
            }
    }
    return dist;
}

struct DSU {
    vector<int> p, sz;
    DSU(int n) : p(n + 1), sz(n + 1, 1) {
        iota(p.begin(), p.end(), 0);
    }
    int find(int x) { return p[x] == x ? x : p[x] = find(p[x]); }
    bool unite(int a, int b) {
        a = find(a); b = find(b);
        if (a == b) return false;
        if (sz[a] < sz[b]) swap(a, b);
        p[b] = a; sz[a] += sz[b];
        return true;
    }
};


vector<int> topo(int n, const vector<vector<int>> &g) {
    vector<int> in(n + 1, 0), order;
    for (int v = 1; v <= n; v++) for (int u : g[v]) in[u]++;
    queue<int> q;
    for (int v = 1; v <= n; v++) if (in[v] == 0) q.push(v);
    while (!q.empty()) {
        int v = q.front(); q.pop();
        order.push_back(v);
        for (int u : g[v]) if (--in[u] == 0) q.push(u);
    }
    return order;          // size < n  =>  cycle
}
